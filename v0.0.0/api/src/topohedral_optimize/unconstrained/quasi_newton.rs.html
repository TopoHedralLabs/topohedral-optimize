25//{{{ enum: UpdateMethod
26#[derive(Copy, Clone)]
27/// Hessian approximation update used by a quasi-Newton method.
28pub enum UpdateMethod {
29    /// BFGS update.
30    BFGS,
31    /// DFP update.
32    DFP,
33}
34//}}}
35//{{{ struct: Options
36#[derive(Clone)]
37/// Options for quasi-Newton minimization.
38pub struct Options {
39    /// Common stopping options.
40    pub uncon_opts: UnconstrainedOptions,
41    /// Line-search algorithm.
42    pub ls_method: LineSearchMethod,
43    /// Hessian update formula.
44    pub method: UpdateMethod,
45    /// Restart interval in iterations.
46    pub restart: u64,
47}

49//{{{ struct: QuasiNewton
50/// Stateful quasi-Newton optimizer.
51pub struct QuasiNewton<'a, F: RealFn + ?Sized> {
52    fcn: &'a mut F,
53    x_init: Vector,
54    norm_grad_fx_init: f64,
55    opts: Options,
56    quadratic_model: QuadraticModel,
57}
58//}}}
59//{{{ impl: QuasiNewton
60impl<'a, F: RealFn + ?Sized> QuasiNewton<'a, F> {
61    #[trace_fn]
62    pub fn new(
63        fcn: &'a mut F,
64        x0: Vector,
65        opts: Options,
66    ) -> Self {
67        let grad_0 = fcn.derivative(&x0);
68        let norm_grad_0 = grad_0.abs_max().unwrap_or(0.0);
69        let n = x0.len();
70        Self {
71            fcn,
72            x_init: x0,
73            norm_grad_fx_init: norm_grad_0,
74            opts,
75            quadratic_model: QuadraticModel::new(n),
76        }
77    }
78
79    #[trace_fn]
80    fn apply_restart(
81        &self,
82        k: u64,
83        grad_fk: &Vector,
84        dir_k: &mut Vector,
85    ) {
86        let needs_restart = k.is_multiple_of(self.opts.restart);
87        let is_increasing = grad_fk.dot(dir_k) >= 0.0;
88        if needs_restart || is_increasing {
89            *dir_k = -grad_fk.clone();
90        }
91    }
92
93    #[trace_fn]
94    fn is_converged(
95        &self,
96        grad_norm: f64,
97    ) -> Option<ConvergedReason> {
98        let rtol = self.opts.uncon_opts.grad_rtol;
99        let rtol_converged = (grad_norm / self.norm_grad_fx_init) < rtol;
100        if rtol_converged {
101            return Some(ConvergedReason::Rtol);
102        }
103        let atol_converged = grad_norm < self.opts.uncon_opts.grad_atol;
104        if atol_converged {
105            return Some(ConvergedReason::Atol);
106        }
107        None
108    }
109
110    #[trace_fn]
111    fn update_hessian(
112        &mut self,
113        xk_prev: &Vector,
114        xk: &Vector,
115        grad_fk_prev: &Vector,
116        grad_fk: &Vector,
117    ) -> bool {
118        match self.opts.method {
119            UpdateMethod::BFGS => {
120                let sk: Vector = (xk - xk_prev).into();
121                let yk: Vector = (grad_fk - grad_fk_prev).into();
122                self.quadratic_model.try_update(&sk, &yk, Inverse)
123            }
124            UpdateMethod::DFP => {
125                todo!()
126            }
127        }
128    }
129
130    #[trace_fn]
131    fn update_direction(
132        &mut self,
133        xk_prev: &Vector,
134        xk: &Vector,
135        grad_fk_prev: &Vector,
136        grad_fk: &Vector,
137    ) -> Vector {
138        if !self.update_hessian(xk_prev, xk, grad_fk_prev, grad_fk) {
139            self.quadratic_model.reset();
140            return -grad_fk.clone();
141        }
142        -self.quadratic_model.inv_hess_k.matmul(grad_fk)
143    }
144
145    fn print_status(
146        &self,
147        _k: u64,
148        current_iter: &IterData,
149    ) {
150        //{{{ trace
151        info!(target: "qn", "======================================================================== i = {_k}");
152        trace!(target: "qn", "Current solution: {}", current_iter.x.clone().transpose());
153        info!(target: "qn", "Current values: {current_iter}");
154        info!(target: "qn","Convergence measures:");
155        let stationarity = current_iter.grad_fx.abs_max().unwrap_or(0.0);
156        let _grad_ratio = stationarity / self.norm_grad_fx_init;
157        info!(target: "qn", "||∇f(k)|| / ||∇f(0)|| = {_grad_ratio:.4e}");
158        //}}}
159    }
160}
161//}}}
162//{{{ impl: Minimizer for QuasiNewton
163impl<F: RealFn + ?Sized> Minimizer for QuasiNewton<'_, F> {
164    type Error = Error;
165    type Returns = VectorReturns;
166
167    #[trace_fn]
168    fn minimize(&mut self) -> Result<VectorReturns, Self::Error> {
169        let mut iter_k = IterData::new(&mut *self.fcn, &self.x_init);
170        let mut iter_prev_k = iter_k.clone();
171        iter_prev_k.fx = iter_k.fx + 0.5 * iter_k.norm_grad_fx;
172
173        let mut dir_k = -iter_k.grad_fx.clone();
174        let max_iter = self.opts.uncon_opts.max_iter;
175
176        self.quadratic_model
177            .update_iterate(&iter_k.x, iter_k.fx, &iter_k.grad_fx);
178
179        for k in 1..max_iter {
180            self.print_status(k, &iter_k);
181            self.apply_restart(k, &iter_k.grad_fx, &mut dir_k);
182
183            let alpha_init =
184                ls::initial_step(iter_k.fx, iter_prev_k.fx, iter_k.grad_fx.dot(&dir_k));
185
186            let fx_prev = iter_prev_k.fx;
187            iter_prev_k = iter_k;
188
189            let search_result = ls::search(
190                &mut *self.fcn,
191                &iter_prev_k,
192                &dir_k,
193                alpha_init,
194                self.opts.ls_method.clone(),
195            );
196
197            iter_k = match search_result {
198                Ok(iter) => iter,
199                Err(_) => {
200                    self.quadratic_model.reset();
201                    dir_k = -iter_prev_k.grad_fx.clone();
202                    let alpha_init =
203                        ls::initial_step(iter_prev_k.fx, fx_prev, iter_prev_k.grad_fx.dot(&dir_k));
204                    ls::search(
205                        &mut *self.fcn,
206                        &iter_prev_k,
207                        &dir_k,
208                        alpha_init,
209                        self.opts.ls_method.clone(),
210                    )?
211                }
212            };
213
214            dir_k = self.update_direction(
215                &iter_prev_k.x,
216                &iter_k.x,
217                &iter_prev_k.grad_fx,
218                &iter_k.grad_fx,
219            );
220            self.quadratic_model
221                .update_iterate(&iter_k.x, iter_k.fx, &iter_k.grad_fx);
222
223            let stationarity = iter_k.grad_fx.abs_max().unwrap_or(0.0);
224            if let Some(reason) = self.is_converged(stationarity) {
225                //{{{ trace
226                info!(target: "qn", "=============================================");
227                info!(target: "qn", "Converging with reason {reason:?}");
228                info!(target: "qn","Convergence measures:");
229                let _grad_ratio = stationarity / self.norm_grad_fx_init;
230                info!(target: "qn", "||∇f(k)|| / ||∇f(0)|| = {_grad_ratio:.4e}");
231                info!(target: "qn", "||∇f(k)|| = {:.4e}", stationarity);
232                trace!(target: "qn", "fx = {:.4e} x = {}", iter_k.fx, iter_k.x.clone().transpose());
233                info!(target: "qn", "=============================================");
234                //}}}
235                return Ok(VectorReturns {
236                    fmin: iter_k.fx,
237                    xmin: iter_k.x,
238                    reason,
239                    num_iterations: k as usize,
240                    num_fun_evals: 0,
241                    num_grad_evals: 0,
242                });
243            }
244        }
245        //{{{ trace
246        let _maxiter = self.opts.uncon_opts.max_iter;
247        info!(target: "qn", "Did not converge within {_maxiter} iterations");
248        //}}}
249        Err(Error::MaxIterations(self.opts.uncon_opts.max_iter as usize))
250    }
251}
