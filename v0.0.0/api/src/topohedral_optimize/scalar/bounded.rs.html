19const DEFUALT_XTOL: f64 = 1e-5;
20const DEFUALT_MAX_ITER: usize = 100;
21
22//{{{ struct: Options
23#[derive(Copy, Clone)]
24/// Options for bounded scalar minimization.
25pub struct Options {
26    /// Bounds
27    pub bounds: (f64, f64),
28    /// Absolute tolerance on `x` used as the termination criterion.
29    pub xatol: f64,
30    /// Maximum number of function evaluations.
31    pub max_iter: usize,
32}
33//}}}
34impl Options {
35    /// Creates options for the interval `[lower, upper]`.
36    pub fn new(
37        lower: f64,
38        upper: f64,
39    ) -> Result<Self, ScalarError> {
40        if !lower.is_finite() || !upper.is_finite() {
41            return Err(ScalarError::NonFiniteBounds(lower, upper));
42        }
43        if lower > upper {
44            return Err(ScalarError::InvalidBounds(lower, upper));
45        }
46        Ok(Self {
47            bounds: (lower, upper),
48            xatol: DEFUALT_XTOL,
49            max_iter: DEFUALT_MAX_ITER,
50        })
51    }
52}
53//{{{ struct: Bounded
54/// Bounded minimization of a scalar function over a finite interval `[lower, upper]`.
55pub struct Bounded<F: RealFn1> {
56    fcn: F,
57    opts: Options,
58}
59//}}}
60//{{{ impl: Bounded
61impl<F: RealFn1> Bounded<F> {
62    #[trace_fn]
63    pub fn new(
64        fcn: F,
65        opts: Options,
66    ) -> Self {
67        Self { fcn, opts }
68    }
69}

71//{{{ impl: Minimizer for Bounded
72impl<F: RealFn1> Minimizer for Bounded<F> {
73    type Error = ScalarError;
74    type Returns = ScalarReturns;
75
76    #[trace_fn]
77    fn minimize(&mut self) -> Result<Self::Returns, Self::Error> {
78        let sqrt_eps = 2.2e-16_f64.sqrt();
79        let golden_mean = 0.5 * (3.0 - 5.0_f64.sqrt());
80
81        // `a`/`b` are the shrinking bracket, initially the full bounds.
82        let mut a = self.opts.bounds.0;
83        let mut b = self.opts.bounds.1;
84        let mut fulc = a + golden_mean * (b - a);
85        let mut nfc = fulc;
86        let mut xf = fulc;
87        let mut rat = 0.0_f64;
88        let mut e = 0.0_f64;
89        let mut x = xf;
90        let mut fx = self.fcn.eval(&x);
91        let mut num = 1usize;
92        let mut fu = f64::INFINITY;
93
94        let mut ffulc = fx;
95        let mut fnfc = fx;
96        let mut xm = 0.5 * (a + b);
97        let mut tol1 = sqrt_eps * xf.abs() + self.opts.xatol / 3.0;
98        let mut tol2 = 2.0 * tol1;
99
100        //{{{ trace
101        trace!(target: "scalar", "bounded: searching [{a:.4e}, {b:.4e}], x0 = {xf:.4e}, f0 = {fx:.4e}");
102        //}}}
103
104        while (xf - xm).abs() > (tol2 - 0.5 * (b - a)) {
105            let mut golden = true;
106
107            // Check for parabolic fit.
108            if e.abs() > tol1 {
109                golden = false;
110                let mut r = (xf - nfc) * (fx - ffulc);
111                let mut q = (xf - fulc) * (fx - fnfc);
112                let mut p = (xf - fulc) * q - (xf - nfc) * r;
113                q = 2.0 * (q - r);
114                if q > 0.0 {
115                    p = -p;
116                }
117                q = q.abs();
118                r = e;
119                e = rat;
120
121                // Check for acceptability of parabola.
122                if p.abs() < (0.5 * q * r).abs() && p > q * (a - xf) && p < q * (b - xf) {
123                    rat = p / q;
124                    x = xf + rat;
125
126                    if (x - a) < tol2 || (b - x) < tol2 {
127                        let si = if xm >= xf { 1.0 } else { -1.0 };
128                        rat = tol1 * si;
129                    }
130                } else {
131                    golden = true;
132                }
133            }
134
135            if golden {
136                // Do a golden-section step.
137                e = if xf >= xm { a - xf } else { b - xf };
138                rat = golden_mean * e;
139            }
140
141            let si = if rat >= 0.0 { 1.0 } else { -1.0 };
142            x = xf + si * rat.abs().max(tol1);
143            fu = self.fcn.eval(&x);
144            num += 1;
145
146            if fu <= fx {
147                if x >= xf {
148                    a = xf;
149                } else {
150                    b = xf;
151                }
152                fulc = nfc;
153                ffulc = fnfc;
154                nfc = xf;
155                fnfc = fx;
156                xf = x;
157                fx = fu;
158            } else {
159                if x < xf {
160                    a = x;
161                } else {
162                    b = x;
163                }
164                if fu <= fnfc || nfc == xf {
165                    fulc = nfc;
166                    ffulc = fnfc;
167                    nfc = x;
168                    fnfc = fu;
169                } else if fu <= ffulc || fulc == xf || fulc == nfc {
170                    fulc = x;
171                    ffulc = fu;
172                }
173            }
174
175            xm = 0.5 * (a + b);
176            tol1 = sqrt_eps * xf.abs() + self.opts.xatol / 3.0;
177            tol2 = 2.0 * tol1;
178
179            //{{{ trace
180            trace!(target: "scalar", "bounded: eval {num}, xf = {xf:.4e}, fx = {fx:.4e}");
181            //}}}
182
183            if num >= self.opts.max_iter {
184                //{{{ trace
185                trace!(target: "scalar", "bounded: exceeded max_iter = {}", self.opts.max_iter);
186                //}}}
187                return Err(ScalarError::MaxIterations(self.opts.max_iter));
188            }
189        }
190
191        if xf.is_nan() || fx.is_nan() || fu.is_nan() {
192            //{{{ trace
193            trace!(target: "scalar", "bounded: NaN encountered");
194            //}}}
195            return Err(ScalarError::NaN);
196        }
197
198        //{{{ trace
199        trace!(target: "scalar", "bounded: converged, xmin = {xf:.4e}, fmin = {fx:.4e}, evals = {num}");
200        //}}}
201
202        Ok(ScalarReturns {
203            xmin: xf,
204            fmin: fx,
205            reason: ConvergedReason::Atol,
206            num_iterations: num,
207            num_fun_evals: num,
208            num_grad_evals: 0,
209        })
210    }
211}
