17/// Errors reported by scalar minimization and bracketing.
18#[derive(Error, Debug)]
19pub enum Error {
20    /// The bracketing iteration limit was reached.
21    #[error("Maximum iterations of {0} reached")]
22    MaxIterations(usize),
23    /// Bracketing terminated without a valid triple.
24    #[error(
25        "The algorithm terminated without finding a valid bracket, consider trying different \
26         initial points"
27    )]
28    InvalidBracket,
29    /// A bound was not finite.
30    #[error("Optimization bounds must be finite scalars, got lower = {0}, upper = {1}")]
31    NonFiniteBounds(f64, f64),
32    /// The lower bound exceeded the upper bound.
33    #[error("The lower bound {0} exceeds the upper bound {1}")]
34    InvalidBounds(f64, f64),
35    /// A function evaluation returned NaN.
36    #[error("Function evaluation returned NaN")]
37    NaN,
38    /// Bracket points were not ordered correctly.
39    #[error(
40        "Bracketing values (xa, xb, xc) = ({0}, {1}, {2}) do not fulfill this requirement: \
41         (xa < xb) and (xb < xc)"
42    )]
43    InvalidBracketOrder(f64, f64, f64),
44    /// The middle bracket value was not lower than both endpoints.
45    #[error(
46        "Bracketing values (xa, xb, xc) do not fulfill this requirement: (f(xb) < f(xa)) and \
47         (f(xb) < f(xc))"
48    )]
49    InvalidBracketValues,
50}

52/// Options controlling automatic bracket growth.
53#[derive(Copy, Clone)]
54pub struct BracketOptions {
55    /// Maximum growth factor for a bracket step.
56    pub grow_limit: f64,
57    /// Maximum number of growth steps.
58    pub max_iter: usize,
59}
60
61impl Default for BracketOptions {
62    fn default() -> Self {
63        Self {
64            grow_limit: 110.0,
65            max_iter: 1000,
66        }
67    }
68}

70//{{{ struct: BracketResult
71/// Result of a successful [`bracket`] search.
72///
73/// `xa`, `xb`, `xc` are ordered (either increasing or decreasing) such that `xb`
74/// lies between `xa` and `xc`, and the corresponding function values satisfy
75/// `fb < fa` and `fb < fc`, i.e. `(xa, fa)`, `(xb, fb)`, `(xc, fc)` bracket a
76/// local minimum of the function.
77#[derive(Copy, Clone, Debug, PartialEq)]
78pub struct BracketResult {
79    /// First bracket point.
80    pub xa: f64,
81    /// Middle bracket point.
82    pub xb: f64,
83    /// Last bracket point.
84    pub xc: f64,
85    /// Function value at `xa`.
86    pub fa: f64,
87    /// Function value at `xb`.
88    pub fb: f64,
89    /// Function value at `xc`.
90    pub fc: f64,
91    /// Number of function evaluations.
92    pub num_fun_evals: usize,
93}
94//}}}
95//{{{ fun: bracket
96/// Downhill bracket search for a local minimum.
97///
98/// Given a function `f` and two distinct initial points `a` and `b`, searches
99/// in the downhill direction, as defined by the two initial points, and
100/// returns three points that bracket a local minimum of `f`.
101///
102/// Ported from SciPy's `scipy.optimize.bracket`
103/// (`scipy/optimize/_optimize.py`).
104///
105/// # Errors
106///
107/// - [`Error::MaxIterations`] if `opts.max_iter` growth steps are performed
108///   without finding a bracket.
109/// - [`Error::InvalidBracket`] if the algorithm terminates without satisfying
110///   the validity conditions of a bracket, i.e. `fb < fa`, `fb < fc`, `xb`
111///   strictly between `xa` and `xc`, and all three points finite.
112#[trace_fn]
113pub fn bracket<F: RealFn1>(
114    f: &mut F,
115    a: f64,
116    b: f64,
117    opts: BracketOptions,
118) -> Result<BracketResult, Error> {
119    const GOLD: f64 = 1.618034;
120    const VERY_SMALL_NUM: f64 = 1e-21;
121
122    let mut xa = a;
123    let mut xb = b;
124    let mut fa = f.eval(&xa);
125    let mut fb = f.eval(&xb);
126
127    if fa < fb {
128        std::mem::swap(&mut xa, &mut xb);
129        std::mem::swap(&mut fa, &mut fb);
130    }
131
132    let mut xc = xb + GOLD * (xb - xa);
133    let mut fc = f.eval(&xc);
134    let mut num_fun_evals = 3usize;
135    let mut iter = 0usize;
136
137    //{{{ trace
138    trace!(target: "scalar", "bracket: starting with xa = {xa:.4e}, xb = {xb:.4e}, xc = {xc:.4e}");
139    //}}}
140
141    while fc < fb {
142        let tmp1 = (xb - xa) * (fb - fc);
143        let tmp2 = (xb - xc) * (fb - fa);
144        let val = tmp2 - tmp1;
145        let denom = if val.abs() < VERY_SMALL_NUM {
146            2.0 * VERY_SMALL_NUM
147        } else {
148            2.0 * val
149        };
150        let mut w = xb - ((xb - xc) * tmp2 - (xb - xa) * tmp1) / denom;
151        let wlim = xb + opts.grow_limit * (xc - xb);
152
153        if iter > opts.max_iter {
154            //{{{ trace
155            trace!(target: "scalar", "bracket: exceeded max_iter = {}", opts.max_iter);
156            //}}}
157            return Err(Error::MaxIterations(opts.max_iter));
158        }
159        iter += 1;
160
161        let fw;
162        if (w - xc) * (xb - w) > 0.0 {
163            let mut fw_inner = f.eval(&w);
164            num_fun_evals += 1;
165            if fw_inner < fc {
166                xa = xb;
167                xb = w;
168                fa = fb;
169                fb = fw_inner;
170                break;
171            } else if fw_inner > fb {
172                xc = w;
173                fc = fw_inner;
174                break;
175            }
176            w = xc + GOLD * (xc - xb);
177            fw_inner = f.eval(&w);
178            num_fun_evals += 1;
179            fw = fw_inner;
180        } else if (w - wlim) * (wlim - xc) >= 0.0 {
181            w = wlim;
182            fw = f.eval(&w);
183            num_fun_evals += 1;
184        } else if (w - wlim) * (xc - w) > 0.0 {
185            let mut fw_inner = f.eval(&w);
186            num_fun_evals += 1;
187            if fw_inner < fc {
188                xb = xc;
189                xc = w;
190                w = xc + GOLD * (xc - xb);
191                fb = fc;
192                fc = fw_inner;
193                fw_inner = f.eval(&w);
194                num_fun_evals += 1;
195            }
196            fw = fw_inner;
197        } else {
198            w = xc + GOLD * (xc - xb);
199            fw = f.eval(&w);
200            num_fun_evals += 1;
201        }
202
203        xa = xb;
204        xb = xc;
205        xc = w;
206        fa = fb;
207        fb = fc;
208        fc = fw;
209    }
210
211    let cond1 = (fb < fc && fb <= fa) || (fb < fa && fb <= fc);
212    let cond2 = (xa < xb && xb < xc) || (xc < xb && xb < xa);
213    let cond3 = xa.is_finite() && xb.is_finite() && xc.is_finite();
214
215    if !(cond1 && cond2 && cond3) {
216        //{{{ trace
217        trace!(target: "scalar", "bracket: invalid bracket xa = {xa:.4e}, xb = {xb:.4e}, xc = {xc:.4e}");
218        //}}}
219        return Err(Error::InvalidBracket);
220    }
221
222    //{{{ trace
223    trace!(target: "scalar", "bracket: found xa = {xa:.4e}, xb = {xb:.4e}, xc = {xc:.4e} in {num_fun_evals} evals");
224    //}}}
225
226    Ok(BracketResult {
227        xa,
228        xb,
229        xc,
230        fa,
231        fb,
232        fc,
233        num_fun_evals,
234    })
235}

237//{{{ enum: Bracket
238/// Specifies how the initial bracketing triple is obtained.
239#[derive(Copy, Clone, Default)]
240pub enum Bracket {
241    /// Search for a bracket automatically, starting from the default points `(0, 1)`.
242    #[default]
243    Auto,
244    /// Search for a bracket starting downhill from the two given points.
245    Points(f64, f64),
246    /// Use an explicit three-point bracket `(xa, xb, xc)`.
247    ///
248    /// Validated so that, after ordering `xa` and `xc`, `xb` lies strictly
249    /// between them and `f(xb) < f(xa)` and `f(xb) < f(xc)`.
250    Triple(f64, f64, f64),
251}
252//}}}
253//{{{ fun: resolve_bracket
254/// Resolves a [`Bracket`] specification into a validated [`BracketResult`].
255///
256/// For [`Bracket::Auto`] and [`Bracket::Points`], delegates to [`bracket`]. For
257/// [`Bracket::Triple`], validates the triple directly rather than searching.
258///
259/// # Errors
260///
261/// - [`Error::InvalidBracketOrder`] if, after ordering `xa` and `xc`, `xb`
262///   does not lie strictly between them.
263/// - [`Error::InvalidBracketValues`] if `f(xb)` is not lower than both
264///   `f(xa)` and `f(xc)`.
265/// - Any error from [`bracket`] for the `Auto`/`Points` variants.
266#[trace_fn]
267pub fn resolve_bracket<F: RealFn1>(
268    f: &mut F,
269    spec: Bracket,
270    opts: BracketOptions,
271) -> Result<BracketResult, Error> {
272    match spec {
273        Bracket::Auto => bracket(f, 0.0, 1.0, opts),
274        Bracket::Points(xa, xb) => bracket(f, xa, xb, opts),
275        Bracket::Triple(xa0, xb, xc0) => {
276            let (xa, xc) = if xa0 > xc0 { (xc0, xa0) } else { (xa0, xc0) };
277            if !(xa < xb && xb < xc) {
278                return Err(Error::InvalidBracketOrder(xa, xb, xc));
279            }
280            let fa = f.eval(&xa);
281            let fb = f.eval(&xb);
282            let fc = f.eval(&xc);
283            if !(fb < fa && fb < fc) {
284                return Err(Error::InvalidBracketValues);
285            }
286            Ok(BracketResult {
287                xa,
288                xb,
289                xc,
290                fa,
291                fb,
292                fc,
293                num_fun_evals: 3,
294            })
295        }
296    }
297}

304    struct ScalarFunction<F: Fn(f64) -> f64> {
305        f: F,

315        type Input = f64;
316        type Output = f64;
317        type Derivative = f64;
318        fn dimension_domain(&self) -> usize {
319            1
320        }
321
322        fn dimension_range(&self) -> usize {
323            1
324        }
325        fn eval(
326            &mut self,

329            (self.f)(*x)
330        }

340    fn parabola(x: f64) -> f64 {
341        (x - 1.0).powi(2)
342    }
343
344    fn quartic(x: f64) -> f64 {
345        (x - 2.0).powi(4) + 3.0
346    }
347
348    fn assert_valid_bracket(
349        res: &BracketResult,
350        f: &mut impl RealFn1,
351    ) {
352        assert!(
353            (res.xa < res.xb && res.xb < res.xc) || (res.xc < res.xb && res.xb < res.xa),
354            "xb must lie strictly between xa and xc: xa = {}, xb = {}, xc = {}",
355            res.xa,
356            res.xb,
357            res.xc
358        );
359        assert!(
360            res.fb < res.fa && res.fb < res.fc,
361            "xb must be the lowest point"
362        );
363
364        // cross-check reported function values against the function itself
365        assert!((f.eval(&res.xa) - res.fa).abs() < 1e-12);
366        assert!((f.eval(&res.xb) - res.fb).abs() < 1e-12);
367        assert!((f.eval(&res.xc) - res.fc).abs() < 1e-12);
368    }
369
370    #[test]
371    fn test_bracket_parabola_downhill() {
372        let mut f = ScalarFunction::new(parabola);
373        let res = bracket(&mut f, 0.0, 1.0, BracketOptions::default()).unwrap();
374        assert_valid_bracket(&res, &mut f);
375    }

382        assert_valid_bracket(&res, &mut f);

387        let mut f = ScalarFunction::new(quartic);
388        let res = bracket(&mut f, 0.0, 1.0, BracketOptions::default()).unwrap();
389        assert_valid_bracket(&res, &mut f);
390        assert!(
391            (res.xa < 2.0 && 2.0 < res.xc) || (res.xc < 2.0 && 2.0 < res.xa),
392            "bracket should straddle the true minimum at x = 2: xa = {}, xc = {}",
393            res.xa,
394            res.xc
395        );
396    }
397
398    #[test]
399    fn test_bracket_max_iterations_exceeded() {
400        // A function that is monotonically decreasing everywhere never brackets a
401        // minimum, so with a tiny iteration budget the search should report
402        // MaxIterations rather than loop forever.
403        let mut f = ScalarFunction::new(|x: f64| -x);
404        let opts = BracketOptions {
405            grow_limit: 110.0,
406            max_iter: 2,
407        };
408        let res = bracket(&mut f, 0.0, 1.0, opts);
409        assert!(matches!(res, Err(Error::MaxIterations(2))));
410    }

416        assert!(res.num_fun_evals >= 3);
