20/// Options for the Nocedal line search.
21pub struct Options {
22    /// Shared line-search conditions and step limits.
23    pub ls_opts: com::Options,
24    /// Maximum number of outer iterations.
25    pub maxiter: usize,
26    /// Maximum number of zoom iterations.
27    pub zoom_maxiter: usize,
28}

31pub struct Nocedal<F: RealFn1> {
32    pub opts: Options,
33    pub(crate) f: F,
34}

47//{{{ impl: LineSearch for Nocedal
48impl<F: RealFn1> LineSearch for Nocedal<F> {
49    type Function = F;
50    #[trace_fn]
51    fn search(
52        &mut self,
53        phi0: f64,
54        dphi0: f64,
55        mut alpha1: f64,
56    ) -> Result<Returns, Error> {
57        //{{{ trace
58        trace!(target: "ls", "Entering with phi0 = {:.4e} dphi0 = {:.4e}", phi0, dphi0);
59        //}}}
60
61        if dphi0 > 0.0 {
62            return Err(Error::NotDecreasing);
63        }
64        let c1 = self.opts.ls_opts.c1;
65        let c2 = self.opts.ls_opts.c2;
66        let step_min = self.opts.ls_opts.step_min;
67        let step_max = self.opts.ls_opts.step_max;
68        alpha1 = alpha1.clamp(step_min, step_max);
69        let mut alpha0 = 0.0;
70        let mut phi_a0 = phi0;
71        let mut dphi_a0 = dphi0;
72        let mut dphi_a1 = 0.0;
73        let _ = dphi_a1;
74        let mut phi_a1 = self.f.eval(&alpha1);
75        let max_iter = self.opts.maxiter;
76
77        for i in 0..max_iter {
78            //{{{ trace
79            trace!(target: "ls", "--------------------------------------- nocedal it = {}", i);
80            trace!(target: "ls", "alpha0 = {:.4e} alpha1 = {:.4e}", alpha0, alpha1);
81            trace!(target: "ls", "phi_a0 = {:.4e} phi1 {:.4e}", phi_a0, phi_a1);
82            trace!(target: "ls", "dphi_aa0 = {:.4e} dphi1 {:.4e}", dphi_a0, dphi_a1);
83            //}}}
84
85            if alpha1 < f64::EPSILON * 100.0 {
86                //{{{ trace
87                error!(target: "ls", "Too small step size detected");
88                //}}}
89                return Err(Error::StepSizeSmall);
90            }
91
92            let not_first_iteration = i > 0;
93            let not_decreasing = phi_a1 >= phi_a0;
94
95            // First check if current step is armijo-acceptable, if not then try zoom and check
96            // again.
97            if !satisfies_armijo(c1, alpha1, phi0, dphi0, phi_a1)
98                || (not_decreasing && not_first_iteration)
99            {
100                //{{{ trace
101                trace!(target: "ls", "Does not satisfy armijo");
102                //}}}
103                let zoom_result = zoom(
104                    alpha0,
105                    alpha1,
106                    phi_a0,
107                    phi_a1,
108                    dphi_a0,
109                    &mut self.f,
110                    phi0,
111                    dphi0,
112                    c1,
113                    c2,
114                    self.opts.zoom_maxiter,
115                );
116
117                let (alpha_tmp, phi_tmp, _dphi_tmp) = match zoom_result {
118                    None => {
119                        //{{{ trace
120                        error!(target: "ls","Zoom failed");
121                        //}}}
122                        return Err(Error::NotDecreasing);
123                    }
124                    Some(result) => result,
125                };
126
127                //{{{ trace
128                error!(target: "ls", "Leaving with {:.4e} {:.4e} {:.4e}", alpha_tmp, phi_tmp, _dphi_tmp);
129                //}}}
130                return Ok(Returns {
131                    alpha: alpha_tmp,
132                    phi_alpha: phi_tmp,
133                });
134            }
135
136            // current step is armijo-acceptable, so check if curvature-accepttable
137            dphi_a1 = self.f.derivative(&alpha1);
138            //{{{ trace
139            trace!(target: "ls", "dphi_a1 = {dphi_a1:.4e}");
140            //}}}
141            if dphi_a1.abs() <= -c2 * dphi0 {
142                //{{{ trace
143                trace!(target: "ls", "Satisfies curvature");
144                trace!(target: "ls","Returning alpha = {:.4e} falpha = {:.4e}", alpha1, phi_a1);
145                //}}}
146                return Ok(Returns {
147                    alpha: alpha1,
148                    phi_alpha: phi_a1,
149                });
150            }
151
152            if dphi_a1 >= 0.0 {
153                //{{{ trace
154                trace!(target: "ls", "Curvature is positive {:.4e}", dphi_a1);
155                //}}}
156                let zoom_result = zoom(
157                    alpha1,
158                    alpha0,
159                    phi_a1,
160                    phi_a0,
161                    dphi_a1,
162                    &mut self.f,
163                    phi0,
164                    dphi0,
165                    c1,
166                    c2,
167                    self.opts.zoom_maxiter,
168                );
169                let (alpha_tmp, phi_tmp, _dphi_tmp) = match zoom_result {
170                    None => {
171                        //{{{  trace
172                        error!(target: "ls", "Zoom failed");
173                        //}}}
174                        return Err(Error::NotDecreasing);
175                    }
176                    Some(result) => {
177                        //{{{ trace
178                        trace!(target: "ls","Zoom succeeded with result {:.4e} {:.4e} {:.4e}",
179                                  result.0, result.1, result.2);
180                        //}}}
181                        result
182                    }
183                };
184                //{{{ trace
185                error!(target: "ls", "Returning alpha = {:.4e} falpha = {:.4e}", alpha_tmp, phi_tmp);
186                //}}}
187                return Ok(Returns {
188                    alpha: alpha_tmp,
189                    phi_alpha: phi_tmp,
190                });
191            }
192
193            if alpha1 >= step_max {
194                //{{{ trace
195                trace!(target: "ls", "Reached maximum step size, returning current alpha");
196                //}}}
197                return Ok(Returns {
198                    alpha: alpha1,
199                    phi_alpha: phi_a1,
200                });
201            }
202
203            //{{{ trace
204            trace!(target: "ls", "Doubling alpha for next iteration to {:.4e}", (2.0 * alpha1).min(step_max));
205            //}}}
206            let alpha2 = (2.0 * alpha1).min(step_max);
207            alpha0 = alpha1;
208            alpha1 = alpha2;
209            phi_a0 = phi_a1;
210            dphi_a0 = dphi_a1;
211            phi_a1 = self.f.eval(&alpha1);
212        }
213
214        //{{{ trace
215        error!("Reached max number of iterations");
216        //}}}
217        Ok(Returns {
218            alpha: alpha1,
219            phi_alpha: phi_a1,
220        })
221    }
222}

224//{{{ fun: zoom
225#[allow(clippy::too_many_arguments, clippy::identity_op)]
226#[trace_fn]
227fn zoom<F: RealFn1>(
228    mut a_lo: f64,
229    mut a_hi: f64,
230    mut phi_lo: f64,
231    mut phi_hi: f64,
232    mut dphi_lo: f64,
233    phi_fcn: &mut F,
234    phi0: f64,
235    dphi0: f64,
236    c1: f64,
237    c2: f64,
238    max_iter: usize,
239) -> Option<(f64, f64, f64)>
240where
241{
242    //{{{ trace
243    trace!(target: "ls", "Entering with a_lo = {:.4e} a_hi = {:.4e} phi_lo = {:.4e} phi_hi = {:.4e}", a_lo, a_hi, phi_lo, phi_hi);
244    trace!(target: "ls", "phi0 = {:.4e} dphi0 = {:.4e} c1 = {:.4e} c2 = {:.4e}", phi0, dphi0, c1, c2);
245    //}}}
246    let mut iter = 0;
247    let delta1 = 0.2;
248    let delta2 = 0.1;
249    let mut phi_rec = phi0;
250    let mut a_rec = 0.0;
251    let mut cchk = 0.0;
252    let mut qchk = 0.0;
253    let _ = qchk;
254    loop {
255        //{{{ trace
256        debug!(target: "ls", "...............zoom iter = {}", iter);
257        trace!(target: "ls", "cchk = {:.4e}  qchk = {:.4e}", cchk, qchk);
258        //}}}
259
260        let dalpha = a_hi - a_lo;
261
262        let (a, b) = if dalpha < 0.0 {
263            (a_hi, a_lo)
264        } else {
265            (a_lo, a_hi)
266        };
267
268        //{{{ trace
269        trace!(target: "ls", "dalpha = {:.4e} a_lo = {:.4e} a_hi = {:.4e}", dalpha, a_lo, a_hi);
270        //}}}
271
272        let mut opt_a_j: Option<f64> = None;
273
274        // first try cubic interpolation
275        if iter > 0 {
276            //{{{ trace
277            trace!(target: "ls", "trying cubic interpolation");
278            //}}}
279            opt_a_j = cubicmin3(a_lo, phi_lo, dphi_lo, a_hi, phi_hi, a_rec, phi_rec);
280            cchk = delta1 * dalpha;
281        }
282
283        // if not good enough first try quadratic interpolation
284        if iter == 0
285            || opt_a_j.is_none()
286            || opt_a_j.unwrap() > b - cchk
287            || opt_a_j.unwrap() < a + cchk
288        {
289            //{{{ trace
290            trace!(target: "ls", "trying quadratic interpolation");
291            //}}}
292            qchk = delta2 * dalpha;
293            opt_a_j = quadmin(a_lo, phi_lo, dphi_lo, a_hi, phi_hi);
294
295            // finally try bisection
296            if opt_a_j.is_none() || opt_a_j.unwrap() > b - qchk || opt_a_j.unwrap() < a + qchk {
297                //{{{ trace
298                trace!(target: "ls", "Trying bisection");
299                //}}}
300                opt_a_j = Some(0.5 * (a + b));
301            }
302        }
303
304        let a_j = opt_a_j.unwrap();
305        //{{{ trace
306        trace!(target: "ls", "New value of a_j = {:.4e}", a_j);
307        //}}}
308        // try new value of alpha
309        let phi_aj = phi_fcn.eval(&a_j);
310
311        let not_sat_armijo = !satisfies_armijo(c1, a_j, phi0, dphi0, phi_aj);
312        let not_decreasing = phi_aj >= phi_lo;
313
314        if not_sat_armijo || not_decreasing {
315            //{{{ trace
316            trace!(target: "ls", "Failed armijo condition with {} {}", not_sat_armijo, not_decreasing);
317            //}}}
318            phi_rec = phi_hi;
319            a_rec = a_hi;
320            a_hi = a_j;
321            phi_hi = phi_aj;
322        } else {
323            //{{{ trace
324            trace!(target: "ls", "Passed armijo condition");
325            //}}}
326            let dphi_aj = phi_fcn.derivative(&a_j);
327            if dphi_aj.abs() <= -c2 * dphi0 {
328                //{{{ trace
329                trace!(target: "ls","Passed curvature condition");
330                error!(target: "ls", "Returning a_j = {:.4e} phi_aj = {:.4e} dphi_aj = {:.4e}",
331                      a_j, phi_aj, dphi_aj);
332                //}}}
333                return Some((a_j, phi_aj, dphi_aj));
334            }
335
336            if dphi_aj * (a_hi - a_lo) >= 0.0 {
337                phi_rec = phi_hi;
338                a_rec = a_hi;
339                a_hi = a_lo;
340                phi_hi = phi_lo;
341            } else {
342                phi_rec = phi_lo;
343                a_rec = a_lo;
344            }
345
346            a_lo = a_j;
347            phi_lo = phi_aj;
348            dphi_lo = dphi_aj;
349        }
350        iter += 1;
351        if iter == max_iter {
352            //{{{ trace
353            error!(target: "ls", "Reached max iterations in zoom");
354            //}}}
355            return None;
356        }
357    }
358}

369    struct ScalarFunction<F: Fn(f64) -> f64, G: Fn(f64) -> f64> {
370        f: F,
371        df_dx: G,
372    }
373
374    impl<F: Fn(f64) -> f64, G: Fn(f64) -> f64> ScalarFunction<F, G> {
375        #[trace_fn]
376        pub fn new(
377            f: F,
378            df_dx: G,
379        ) -> Self {
380            Self { f, df_dx }
381        }
382    }
383
384    impl<F: Fn(f64) -> f64, G: Fn(f64) -> f64> crate::DifferentiableFn for ScalarFunction<F, G> {
385        type Input = f64;
386        type Output = f64;
387        type Derivative = f64;
388        #[trace_fn]
389        fn dimension_domain(&self) -> usize {
390            1
391        }
392
393        #[trace_fn]
394        fn dimension_range(&self) -> usize {
395            1
396        }
397        #[trace_fn]
398        fn eval(
399            &mut self,
400            x: &f64,
401        ) -> f64 {
402            (self.f)(*x)
403        }
404
405        #[trace_fn]
406        fn derivative(
407            &mut self,
408            x: &f64,
409        ) -> f64 {
410            (self.df_dx)(*x)
411        }
412    }

414    //{{{ collection: zoom tests
415    #[test]
416    #[trace_fn]
417    fn test_zoom_quad_left() {
418        let f = |x: f64| (x - 2.0).powi(2);
419        let df_dx = |x: f64| 2.0 * (x - 2.0);
420        let mut sf = ScalarFunction::new(f, df_dx);
421        let phi0 = sf.eval(&0.0);
422        let dphi0 = sf.derivative(&0.0);
423        let phi1 = sf.eval(&1.0);
424        let res = zoom(
425            0.0, 1.0, phi0, phi1, dphi0, &mut sf, phi0, dphi0, 1e-4, 0.9, 10,
426        );
427
428        assert!(res.is_some());
429        let res = res.unwrap();
430        assert_relative_eq!(res.0, 0.5, epsilon = 1e-6);
431        assert_relative_eq!(res.1, 2.25, epsilon = 1e-6);
432        assert_relative_eq!(res.2, -3.0, epsilon = 1e-6);
433    }
