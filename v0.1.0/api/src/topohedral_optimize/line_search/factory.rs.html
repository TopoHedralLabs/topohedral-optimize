19//{{{ enum: Method
20/// Selects a line-search algorithm.
21#[cfg_attr(feature = "serde", derive(serde::Deserialize, serde::Serialize))]
22#[derive(Clone, Debug, PartialEq)]
23#[non_exhaustive]
24pub enum Method {
25    /// More-Thuente line search.
26    Thuente(thuente::Options),
27    /// Nocedal line search.
28    Nocedal(nocedal::Options),
29}
30//}}}
31impl Method {
32    /// Validates the selected algorithm's options.
33    ///
34    /// # Errors
35    ///
36    /// Returns [`ValidationError`] if any option is invalid.
37    pub fn validate(&self) -> Result<(), ValidationError> {
38        match self {
39            Self::Thuente(options) => options.validate(),
40            Self::Nocedal(options) => options.validate(),
41        }
42    }
43}
44//{{{ fun: create
45#[trace_fn]
46pub fn create<'a, F: RealFn1 + 'a>(
47    fcn: F,
48    method: Method,
49) -> Box<dyn LineSearch<Function = F> + 'a> {
50    match method {
51        Method::Thuente(opts) => Box::new(thuente::Thuente::new(fcn, opts)),
52        Method::Nocedal(opts) => Box::new(nocedal::Nocedal::new(fcn, opts)),
53    }
54}
55//}}}
