231    /// Validates this configuration.
232    ///
233    /// # Errors
234    ///
235    /// Returns [`ValidationError`] if a nested method is invalid, an iteration
236    /// counter is zero, a tuning parameter is outside `(0, 1)`, or the step
237    /// interval is invalid.
238    pub fn validate(&self) -> Result<(), ValidationError> {
239        self.bound_opts.validate()?;
240        self.unconstrained_method.validate()?;
241        validate_positive_finite("mu", self.mu)?;
242        validate_unit_interval("rho", self.rho)?;
243        validate_nonzero("n1", self.n1 as u64)?;
244        validate_nonzero("n2", self.n2 as u64)?;
245        validate_nonzero("memory", self.memory as u64)?;
246        validate_unit_interval("delta", self.delta)?;
247        validate_unit_interval("eta", self.eta)?;
248        validate_positive_finite("alpha_min", self.alpha_min)?;
249        validate_positive_finite("alpha_max", self.alpha_max)?;
250        if self.alpha_min >= self.alpha_max {
251            return Err(ValidationError::InvalidFloat {
252                parameter: "alpha_min",
253                value: self.alpha_min,
254                requirement: "must be less than alpha_max",
255            });
256        }
257        Ok(())
258    }
259}

261fn validate_unit_interval(
262    parameter: &'static str,
263    value: f64,
264) -> Result<(), ValidationError> {
265    if !value.is_finite() || value <= 0.0 || value >= 1.0 {
266        return Err(ValidationError::InvalidFloat {
267            parameter,
268            value,
269            requirement: "must be finite and strictly between zero and one",
270        });
271    }
272    Ok(())
273}

361//{{{ impl: ActiveSetAlgorithm
362impl<'a, F: RealFn + ?Sized> ActiveSetAlgorithm<'a, F> {
363    #[trace_fn]
364    pub fn new(
365        fcn: &'a mut F,
366        bounds: BoundConstraints,
367        mut x0: Vector,
368        opts: Options,
369    ) -> Self {
370        bounds.clamp(&mut x0);
371        let grad_0 = fcn.derivative(&x0);
372        let negative_grad_0 = -grad_0.clone();
373        let projectd_grad_0 = bounds.projected_direction(&x0, &negative_grad_0, 1.0);
374        let norm_projected_grad_0 = projectd_grad_0.abs_max().unwrap();
375
376        let n1 = opts.n1;
377        let m = opts.memory;
378
379        //{{{ trace
380        info!(target: "asa", "Initializing active-set algorithm");
381        trace!(target: "asa", "Initial solution: {}", x0.clone().transpose());
382        info!(target: "asa", "Initial ||∇f_proj||_∞ = {norm_projected_grad_0:.4e}");
383        trace!(target: "asa", "Active-set history length = {n1}, function history length = {m}");
384        //}}}
385
386        Self {
387            fcn,
388            bounds,
389            x_init: x0,
390            norm_grad_fx_init: norm_projected_grad_0,
391            opts,
392            fn_history: CircularBuffer::new(m),
393            active_signature_history: CircularBuffer::new(n1),
394        }
395    }
396
397    #[trace_fn]
398    fn is_converged(
399        &self,
400        iter_k: &IterData,
401    ) -> Option<ConvergedReason> {
402        //{{{ trace
403        trace!(target: "asa", "Checking convergence");
404        //}}}
405        let projected_grad =
406            self.bounds
407                .projected_direction(&iter_k.x, &(-iter_k.grad_fx.clone()), 1.0);
408        let projected_grad_norm = projected_grad.abs_max().unwrap_or(0.0);
409        let _projected_grad_ratio = if self.norm_grad_fx_init > 0.0 {
410            projected_grad_norm / self.norm_grad_fx_init
411        } else {
412            0.0
413        };
414        let rtol_reached =
415            projected_grad_norm < self.opts.bound_opts.base_opts.grad_rtol * self.norm_grad_fx_init;
416
417        //{{{ trace
418        trace!(target: "asa",
419            "||∇f_proj|| / ||∇f_proj(0)|| = {_projected_grad_ratio:.4e}, ||∇f_proj|| = {projected_grad_norm:.4e}"
420        );
421        //}}}
422
423        if rtol_reached {
424            //{{{ trace
425            trace!(target: "asa", "Rtol reached");
426            //}}}
427            return Some(ConvergedReason::Rtol);
428        }
429
430        let atol_reached = projected_grad_norm < self.opts.bound_opts.base_opts.grad_atol;
431
432        if atol_reached {
433            //{{{ trace
434            trace!(target: "asa", "Atol reached");
435            //}}}
436            return Some(ConvergedReason::Atol);
437        }
438
439        None
440    }

442    #[trace_fn]
443    fn bb_step(
444        &mut self,
445        s: &Vector,
446        y: &Vector,
447        fallback: f64,
448    ) -> f64 {
449        let s_dot_y = s.dot(y);
450        if s_dot_y <= 0.0 {
451            //{{{ trace
452            debug!(target: "asa", "Skipping BB update because sᵀy = {s_dot_y:.4e}; fallback alpha = {fallback:.4e}");
453            //}}}
454            return fallback;
455        }
456        let a = s.dot(s) / s_dot_y;
457        let alpha = a.clamp(self.opts.alpha_min, self.opts.alpha_max);
458        //{{{ trace
459        debug!(target: "asa", "BB update: sᵀy = {s_dot_y:.4e}, raw alpha = {a:.4e}, clamped alpha = {alpha:.4e}");
460        //}}}
461        alpha
462    }

464    #[trace_fn]
465    fn ngpa_step(
466        &mut self,
467        iter_k: &IterData,
468        alpha_init: f64,
469    ) -> Option<IterData> {
470        let IterData {
471            x,
472            fx,
473            grad_fx,
474            norm_grad_fx: _,
475        } = iter_k;
476
477        let d = self
478            .bounds
479            .projected_direction(x, &(-grad_fx.clone()), alpha_init);
480        let _d_norm = d.norm();
481        let _d_inf_norm = d.abs_max().unwrap_or(0.0);
482
483        //{{{ trace
484        debug!(target: "asa", "NGPA step: alpha_init = {alpha_init:.4e}, ||d|| = {_d_norm:.4e}, ||d||_∞ = {_d_inf_norm:.4e}");
485        //}}}
486
487        if d.abs_max().unwrap() < SMALL {
488            //{{{ trace
489            debug!(target: "asa", "Projected direction is below SMALL = {SMALL:.4e}");
490            //}}}
491            return None;
492        }
493
494        let f_max = self.fn_history.max().unwrap_or(*fx);
495        let delta = self.opts.delta;
496        let mut alpha = 1.0;
497        let mut x_trial: Vector = (x + &d).into();
498        let mut f_trial = self.fcn.eval(&x_trial);
499        let gradfk_dot_d = grad_fx.dot(&d);
500
501        //{{{ trace
502        trace!(target: "asa", "Running nonmonotone Armijo search: f_max = {f_max:.4e}, gᵀd = {gradfk_dot_d:.4e}, delta = {delta:.4e}");
503        //}}}
504        let max_iterations = 25;
505        for _i in 0..max_iterations {
506            let armijo_rhs = f_max + delta * alpha * gradfk_dot_d;
507            trace!(target: "asa", "Armijo trial {_i}: alpha = {alpha:.4e}, f_trial = {f_trial:.4e}, rhs = {armijo_rhs:.4e}");
508            if f_trial < armijo_rhs || alpha < SMALL {
509                //{{{ trace
510                debug!(target: "asa", "Accepted NGPA step at trial {_i}: alpha = {alpha:.4e}, f_trial = {f_trial:.4e}");
511                //}}}
512                break;
513            }
514            alpha *= self.opts.eta;
515            x_trial = (x + alpha * &d).into();
516            f_trial = self.fcn.eval(&x_trial)
517        }
518
519        let grad_fx_new = self.fcn.derivative(&x_trial);
520        let norm_grad_fx_new = grad_fx_new.abs_max().unwrap_or(0.0);
521        //{{{ trace
522        debug!(target: "asa", "NGPA accepted point: f = {f_trial:.4e}, ||∇f|| = {norm_grad_fx_new:.4e}");
523        //}}}
524        Some(IterData {
525            x: x_trial,
526            fx: f_trial,
527            grad_fx: grad_fx_new,
528            norm_grad_fx: norm_grad_fx_new,
529        })
530    }

532    #[trace_fn]
533    fn undecided_set_is_empy(
534        &self,
535        x: &Vector,
536        inactive_grad: &Vector,
537        full_grad_norm: f64,
538    ) -> bool {
539        if full_grad_norm < SMALL {
540            //{{{ trace
541            trace!(target: "asa", "Undecided set empty because projected gradient norm is below SMALL");
542            //}}}
543            return true;
544        }
545
546        let thresh_g = full_grad_norm.powf(ALPHA);
547        let thresh_x = full_grad_norm.powf(BETA);
548        let distances = self.bounds.all_distances(x);
549
550        //{{{ trace
551        trace!(target: "asa", "Checking undecided set: thresh_g = {thresh_g:.4e}, thresh_x = {thresh_x:.4e}");
552        //}}}
553
554        for (_i, (gi, di)) in inactive_grad.iter().zip(distances.iter()).enumerate() {
555            if gi.abs() >= thresh_g && *di > thresh_x {
556                //{{{ trace
557                debug!(target: "asa", "Undecided variable {_i}: |g_i| = {:.4e}, distance = {:.4e}", gi.abs(), di);
558                //}}}
559                return false;
560            }
561        }
562        //{{{ trace
563        trace!(target: "asa", "Undecided set is empty");
564        //}}}
565        true
566    }

568    #[trace_fn]
569    fn active_sets_are_stable(&self) -> bool {
570        if self.active_signature_history.len() < self.opts.n1 {
571            //{{{ trace
572            trace!(
573                target: "asa",
574                "Active-set history not full yet: len = {}, required = {}",
575                self.active_signature_history.len(),
576                self.opts.n1
577            );
578            //}}}
579            return false;
580        }
581        let mut all_equal = true;
582        let most_recent_sig = self.active_signature_history.newest().unwrap().clone();
583        for sig in self.active_signature_history.iter() {
584            all_equal = all_equal && (*sig == most_recent_sig);
585        }
586        //{{{ trace
587        debug!(target: "asa", "Active-set stability over last {} signatures: {all_equal}", self.opts.n1);
588        //}}}
589        return all_equal;
590    }

610        info!(target: "asa", "Convergence measures:");

612        info!(target: "asa", "||∇f_proj(k)|| = {projected_grad_norm:.4e}");
613        trace!(target: "asa", "∇f_proj: {}", projected_grad.transpose());
614    }

617//{{{ impl: Minimizer for ActiveSetAlgorithm
618impl<F: RealFn + ?Sized> Minimizer for ActiveSetAlgorithm<'_, F> {
619    type Error = super::common::Error;
620    type Returns = crate::VectorReturns;
621
622    #[trace_fn]
623    fn minimize(&mut self) -> Result<crate::VectorReturns, Self::Error> {
624        let mut iter_k_prev = IterData::new(&mut *self.fcn, &self.x_init);
625        let mut iter_k = IterData::new(&mut *self.fcn, &self.x_init);
626        let mut phase = Phase::Ngpa;
627        let mut mu = self.opts.mu;
628        let mut alpha_bb = 1.0;
629
630        self.fn_history.append(iter_k.fx);
631        self.active_signature_history
632            .append(self.bounds.active_signature(&iter_k.x));
633
634        info!(target: "asa", "Starting active-set iterations with phase = {phase:?}, mu = {mu:.4e}, alpha_bb = {alpha_bb:.4e}");
635
636        for k in 0..self.opts.bound_opts.base_opts.max_iter {
637            self.print_status(k, &iter_k);
638            if let Some(reason) = self.is_converged(&iter_k) {
639                //{{{ trace
640                info!(target: "asa", "=============================================");
641                info!(target: "asa", "Converging with reason {reason:?}");
642                trace!(target: "asa", "fx = {:.4e} x = {}", iter_k.fx, iter_k.x.clone().transpose());
643                info!(target: "asa", "=============================================");
644                //}}}
645                return Ok(crate::VectorReturns {
646                    xmin: iter_k.x,
647                    fmin: iter_k.fx,
648                    reason,
649                    num_iterations: k as usize,
650                    num_fun_evals: 0,
651                    num_grad_evals: 0,
652                });
653            }
654
655            match phase {
656                Phase::Ngpa => {
657                    //{{{ trace
658                    debug!(target: "asa", "Entering NGPA phase with alpha_bb = {alpha_bb:.4e}, mu = {mu:.4e}");
659                    //}}}
660                    iter_k_prev.copy_from(&iter_k);
661
662                    let IterData {
663                        x: x_prev,
664                        fx: _,
665                        grad_fx: grad_fx_prev,
666                        norm_grad_fx: _,
667                    } = iter_k_prev.clone();
668
669                    let npga_ok = self.ngpa_step(&iter_k, alpha_bb);
670
671                    if npga_ok.is_none() {
672                        //{{{ trace
673                        debug!(target: "asa", "NGPA could not produce a step; terminating iteration loop");
674                        //}}}
675                        break;
676                    }
677
678                    iter_k.copy_from(&npga_ok.unwrap());
679
680                    let IterData {
681                        x,
682                        fx,
683                        grad_fx,
684                        norm_grad_fx: _,
685                    } = &iter_k;
686
687                    self.fn_history.append(*fx);
688                    self.active_signature_history
689                        .append(self.bounds.active_signature(x));
690
691                    let s = (x - x_prev).into();
692                    let y = (grad_fx - grad_fx_prev).into();
693                    alpha_bb = self.bb_step(&s, &y, alpha_bb);
694                    //{{{ trace
695                    trace!(target: "asa", "alpha_bb = {alpha_bb:.4e}");
696                    //}}}
697
698                    let projected_grad =
699                        self.bounds.projected_direction(x, &(-grad_fx.clone()), 1.0);
700                    let projected_grad_norm = projected_grad.norm();
701                    let inactive_grad = self.bounds.masked_gradient(x, grad_fx);
702                    let inactive_grad_norm = inactive_grad.norm();
703                    let _switch_threshold = mu * projected_grad_norm;
704
705                    //{{{ trace
706                    debug!(
707                        target: "asa",
708                        "NGPA metrics: ||∇f_proj|| = {projected_grad_norm:.4e}, ||∇f_inactive|| = {inactive_grad_norm:.4e}, mu ||∇f_proj|| = {_switch_threshold:.4e}"
709                    );
710                    //}}}
711
712                    if self.undecided_set_is_empy(x, &inactive_grad, projected_grad_norm) {
713                        if inactive_grad_norm < mu * projected_grad_norm {
714                            let _old_mu = mu;
715                            mu *= self.opts.rho;
716                            //{{{ trace
717                            debug!(target: "asa", "Undecided set empty and inactive gradient is small; shrinking mu from {_old_mu:.4e} to {mu:.4e}");
718                            //}}}
719                        } else {
720                            //{{{ trace
721                            debug!(target: "asa", "Undecided set empty but inactive gradient is large; switching to UA");
722                            //}}}
723                            phase = Ua;
724                        }
725                    }
726
727                    if self.active_sets_are_stable()
728                        && inactive_grad_norm >= mu * projected_grad_norm
729                    {
730                        //{{{ trace
731                        debug!(target: "asa", "Active set is stable and inactive gradient is large; switching to UA");
732                        //}}}
733                        phase = Ua;
734                    }
735                }
736                Phase::Ua => {
737                    //{{{ trace
738                    debug!(target: "asa", "Entering UA phase with mu = {mu:.4e}");
739                    //}}}
740                    iter_k_prev.copy_from(&iter_k);
741
742                    let IterData {
743                        x,
744                        fx: _,
745                        grad_fx: _,
746                        norm_grad_fx: _,
747                    } = &iter_k;
748
749                    let active_count_before = self
750                        .bounds
751                        .bound_statuses(x, None)
752                        .iter()
753                        .filter(|status| **status != BoundStatus::Free)
754                        .count();
755
756                    let bounds_statuses = self.bounds.bound_statuses(x, None);
757                    let _free_count = bounds_statuses
758                        .iter()
759                        .filter(|status| **status == BoundStatus::Free)
760                        .count();
761                    //{{{ trace
762                    debug!(target: "asa", "UA restricted problem: active_count = {active_count_before}, free_count = {_free_count}");
763                    //}}}
764                    let x0 = restrict(&bounds_statuses, x);
765                    if x0.is_empty() {
766                        //{{{ trace
767                        debug!(target: "asa", "UA restricted problem has no free variables; switching to NGPA");
768                        //}}}
769                        phase = Phase::Ngpa;
770                        continue;
771                    }
772
773                    let mut restricted_fcn =
774                        RestrictedFunction::new(&mut *self.fcn, x, bounds_statuses.clone());
775                    let res = minimize_impl(
776                        &mut restricted_fcn,
777                        x0,
778                        self.opts.unconstrained_method.clone(),
779                    );
780                    drop(restricted_fcn);
781
782                    let Ok(res) = res else {
783                        //{{{ trace
784                        debug!(target: "asa", "UA internal minimization failed; switching to NGPA");
785                        //}}}
786                        phase = Phase::Ngpa;
787                        continue;
788                    };
789
790                    //{{{ trace
791                    debug!(
792                        target: "asa",
793                        "UA internal minimization returned f = {:.4e}, reason = {:?}, iterations = {}",
794                        res.fmin,
795                        res.reason,
796                        res.num_iterations
797                    );
798                    //}}}
799
800                    iter_k.x.copy_from(lift(&bounds_statuses, &res.xmin));
801                    self.bounds.clamp(&mut iter_k.x);
802                    iter_k.fx = res.fmin;
803                    iter_k.grad_fx.copy_from(self.fcn.derivative(&iter_k.x));
804                    iter_k.fx = self.fcn.eval(&iter_k.x);
805                    iter_k.norm_grad_fx = iter_k.grad_fx.abs_max().unwrap_or(0.0);
806
807                    let projected_grad_new =
808                        self.bounds
809                            .projected_direction(&iter_k.x, &(-iter_k.grad_fx.clone()), 1.0);
810                    let projected_grad_norm_new = projected_grad_new.norm();
811                    let inactive_grad_new = self.bounds.masked_gradient(&iter_k.x, &iter_k.grad_fx);
812                    let inactive_grad_norm_new = inactive_grad_new.norm();
813                    let active_count_after = self
814                        .bounds
815                        .bound_statuses(&iter_k.x, None)
816                        .iter()
817                        .filter(|status| **status != BoundStatus::Free)
818                        .count();
819                    let _switch_threshold_new = mu * projected_grad_norm_new;
820
821                    //{{{ trace
822                    debug!(
823                        target: "asa",
824                        "UA metrics: ||∇f_proj|| = {projected_grad_norm_new:.4e}, ||∇f_inactive|| = {inactive_grad_norm_new:.4e}, mu ||∇f_proj|| = {_switch_threshold_new:.4e}, active_count_before = {active_count_before}, active_count_after = {active_count_after}"
825                    );
826                    //}}}
827
828                    self.fn_history.append(iter_k.fx);
829                    self.active_signature_history
830                        .append(self.bounds.active_signature(&iter_k.x));
831
832                    if inactive_grad_norm_new < mu * projected_grad_norm_new {
833                        //{{{ trace
834                        debug!(target: "asa", "||∇f_inactive|| < mu ||∇f_proj||; switching to NGPA");
835                        //}}}
836                        phase = Phase::Ngpa;
837                    } else if active_count_after > active_count_before
838                        && active_count_after <= active_count_before + self.opts.n2
839                        && !self.undecided_set_is_empy(
840                            &iter_k.x,
841                            &inactive_grad_new,
842                            projected_grad_norm_new,
843                        )
844                    {
845                        //{{{ trace
846                        debug!(target: "asa", "Active bound count increased within n2 and undecided set remains nonempty; switching to NGPA");
847                        //}}}
848                        phase = Phase::Ngpa;
849                    } else {
850                        //{{{ trace
851                        debug!(target: "asa", "Continuing in UA");
852                        //}}}
853                        phase = Phase::Ua;
854                    }
855                }
856            }
857        }
858        info!(
859            target: "asa",
860            "Finished ASA loop without explicit convergence; returning final iterate after {} iterations",
861            self.opts.bound_opts.base_opts.max_iter
862        );
863        Ok(crate::VectorReturns {
864            xmin: iter_k.x,
865            fmin: iter_k.fx,
866            reason: ConvergedReason::Atol,
867            num_iterations: self.opts.bound_opts.base_opts.max_iter as usize,
868            num_fun_evals: 0,
869            num_grad_evals: 0,
870        })
871    }
872}
