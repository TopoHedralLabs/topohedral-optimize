31//{{{ collection: constants
32const DEFAULT_INITIAL_PENALTY: f64 = 1.0;
33const DEFAULT_CONSTRAINT_IMPROVEMENT_FACTOR: f64 = 0.9;
34const DEFAULT_PENALTY_GROWTH_FACTOR: f64 = 2.5;
35/// Ceiling for ω_k (inner stationarity tolerance) on the very first outer
36/// iteration — matches the old `set_inner_tolerances` atol upper clamp.
37/// From the second outer iteration on, the previous ω_k becomes the ceiling
38/// instead, which is what makes the sequence monotone non-increasing.
39const OMEGA_INIT_CEIL: f64 = 1e-2;

187fn validate_unit_interval(
188    parameter: &'static str,
189    value: f64,
190) -> Result<(), ValidationError> {
191    if !value.is_finite() || value <= 0.0 || value >= 1.0 {
192        return Err(ValidationError::InvalidFloat {
193            parameter,
194            value,
195            requirement: "must be finite and strictly between zero and one",
196        });
197    }

201//{{{ struct: LagrangianPenaltyData
202#[derive(Debug)]
203struct LagrangianPenaltyData<F: RealVectorFn> {
204    pub function: F,
205    pub penalties: Vector,
206    pub shifts: Vector,
207    pub values: Vector,
208    pub gradients: Matrix,
209    pub max_violations: Vec<(f64, bool)>,
210}
211//}}}
212//{{{ impl: LagrangianPenaltyData
213impl<F: RealVectorFn> LagrangianPenaltyData<F> {
214    //{{{ fn: new
215    #[trace_fn]
216    pub fn new(
217        fcn: F,
218        initial_penalty: f64,
219    ) -> Self {
220        let num_constraints = fcn.dimension_range();
221        let dimension = fcn.dimension_domain();
222        //{{{ trace
223        trace!(target: "aug", "Creating constraint data object with {num_constraints} constraints");
224        //}}}
225
226        let initial_penalties = Vector::from_value_vec(initial_penalty, num_constraints, Col);
227        let zero_vector = Vector::zeros_vec(num_constraints, Col);
228        let zero_matrix = Matrix::zeros(dimension, num_constraints);
229
230        Self {
231            function: fcn,
232            penalties: initial_penalties,
233            shifts: zero_vector.clone(),
234            values: zero_vector,
235            gradients: zero_matrix,
236            max_violations: vec![(0.0, false); num_constraints],
237        }
238    }
239    //}}}
240    //{{{ fn: update_values
241    #[trace_fn]
242    fn update_values(
243        &mut self,
244        x: &Vector,
245    ) {
246        self.values = self.function.eval(x);
247    }
248    //}}}
249    //{{{ fn: update_gradients
250    #[trace_fn]
251    fn update_gradients(
252        &mut self,
253        x: &Vector,
254    ) {
255        self.gradients = self.function.derivative(x);
256    }
257    //}}}
258    //{{{ fn: update_max_violations
259    #[trace_fn]
260    fn update_max_violations(
261        &mut self,
262        improvement_factor: f64,
263        is_ineq: bool,
264    ) {
265        let current_max_violations = &mut self.max_violations;
266        let current_values = &self.values;
267
268        //{{{ trace
269        trace!(target: "aug", "current_max_violations = {:?}", current_max_violations);
270        trace!(target: "aug", "current_values= {}", current_values.clone().transpose());
271        //}}}
272
273        for (_i, max_violation, was_violated, constraint_value_i) in current_max_violations
274            .iter_mut()
275            .zip(current_values.iter())
276            .enumerate()
277            .map(|(_i, ((max_violation, was_violated), hi))| (_i, max_violation, was_violated, hi))
278        {
279            let max_violation_val = *max_violation;
280            let constraint_val_tmp = if is_ineq {
281                constraint_value_i.max(0.0)
282            } else {
283                constraint_value_i.abs()
284            };
285            *was_violated = constraint_val_tmp > improvement_factor * max_violation_val;
286            if *was_violated {
287                *max_violation = constraint_value_i.abs()
288            }
289            //{{{ trace
290            info!(target: "aug", "i = {} was_violated = {} max_violation = {:.4e}",
291               _i, was_violated, max_violation);
292            //}}}
293        }
294    }
295    //}}}
296    //{{{ fn: compute_lagrange_multiplier_estimates
297    #[trace_fn]
298    fn compute_lagrange_multiplier_estimates(&self) -> Vector {
299        (&self.penalties * &self.shifts).into()
300    }
301    //}}}
302}

306#[derive(Debug)]

308    data: LagrangianPenaltyData<F>,
309    lagrangian_type: LagrangianType,
310}
311//}}}
312//{{{ impl EqPenalty
313impl<F: RealVectorFn> EqPenalty<F> {
314    //{{{ fn: new
315    #[trace_fn]
316    fn new(
317        fcn: F,
318        initial_penalty: f64,
319        lagrangian_type: LagrangianType,
320    ) -> Self {
321        Self {
322            data: LagrangianPenaltyData::new(fcn, initial_penalty),
323            lagrangian_type,
324        }
325    }
326    //}}}
327    //{{{ fn: update_penalties_shifts
328    #[trace_fn]
329    fn update_penalties_shifts(
330        &mut self,
331        improvement_factor: f64,
332        penalty_increase_factor: f64,
333    ) {
334        self.data.update_max_violations(improvement_factor, false);
335
336        for (i, (_, was_violated)) in self.data.max_violations.iter().enumerate() {
337            if *was_violated {
338                self.data.penalties[i] *= penalty_increase_factor;
339                self.data.shifts[i] /= penalty_increase_factor;
340            } else {
341                let hi = self.data.values[i];
342                self.data.shifts[i] += hi;
343            }
344        }
345    }
346    //}}}
347}

349//{{{ impl RealFn for EqPenalty
350impl<F: RealVectorFn> crate::DifferentiableFn for EqPenalty<F> {
351    type Input = Vector;
352    type Output = f64;
353    type Derivative = Vector;
354    //{{{ fn: dimension_domain
355    #[trace_fn]
356    fn dimension_domain(&self) -> usize {
357        self.data.function.dimension_domain()
358    }
359
360    #[trace_fn]
361    fn dimension_range(&self) -> usize {
362        1
363    }
364    //}}}
365    //{{{ fn: eval
366    #[trace_fn]
367    fn eval(
368        &mut self,
369        x: &Vector,
370    ) -> f64 {
371        self.data.update_values(x);
372        let constraint_value = match self.lagrangian_type {
373            LagrangianType::AugmentedLagrangian => {
374                //{{{ trace
375                trace!(target: "aug", "Computing augmented lagrangian");
376                //}}}
377
378                let mut shifted_h: Vector = (&self.data.values + &self.data.shifts).into();
379                shifted_h.transform(|hi| hi.powi(2));
380                0.5 * self.data.penalties.dot(&shifted_h)
381            }
382            LagrangianType::Lagrangian => {
383                //{{{ trace
384                trace!(target: "aug", "Computing classical lagrangian");
385                //}}}
386                let lambda = self.data.compute_lagrange_multiplier_estimates();
387                lambda.dot(&self.data.values)
388            }
389        };
390        //{{{ trace
391        trace!(target: "aug", "Evaluated equality constraint value: {constraint_value:.4e}");
392        //}}}
393        constraint_value
394    }
395    //}}}
396    //{{{ fn: grad
397    #[trace_fn]
398    fn derivative(
399        &mut self,
400        x: &Vector,
401    ) -> Vector {
402        self.data.update_gradients(x);
403        let weighted_constraint_values: Vector = match self.lagrangian_type {
404            LagrangianType::AugmentedLagrangian => {
405                (&self.data.penalties * (&self.data.values + &self.data.shifts)).into()
406            }
407            LagrangianType::Lagrangian => self.data.compute_lagrange_multiplier_estimates(),
408        };
409        self.data.gradients.matmul(&weighted_constraint_values)
410    }
411    //}}}
412}

424    //{{{ fn: new
425    #[trace_fn]
426    fn new(
427        fcn: F,
428        initial_penalty: f64,
429        lagrangian_type: LagrangianType,
430    ) -> Self {
431        Self {
432            data: LagrangianPenaltyData::new(fcn, initial_penalty),
433            lagrangian_type,
434        }
435    }
436    //}}}
437    //{{{ fn: update_penalties_shifts
438    #[trace_fn]
439    fn update_penalties_shifts(
440        &mut self,
441        improvement_factor: f64,
442        penalty_increase_factor: f64,
443    ) {
444        self.data.update_max_violations(improvement_factor, true);
445
446        for (i, (_, was_violated)) in self.data.max_violations.iter().enumerate() {
447            //{{{ trace
448            debug!(target: "aug", "i = {i} was_violated = {was_violated}");
449            //}}}
450            if *was_violated {
451                //{{{ trace
452                debug!(target: "aug", "i = {i} Constraint not improved, increasing penalties");
453                //}}}
454                self.data.penalties[i] *= penalty_increase_factor;
455                self.data.shifts[i] /= penalty_increase_factor;
456            } else {
457                let gi = self.data.values[i];
458                let old_shift = self.data.shifts[i];
459                self.data.shifts[i] = f64::max(0.0, old_shift + gi);
460
461                //{{{ trace
462                debug!(target: "aug", "i = {i} Constraint improved, updating shifts: old_shift = {old_shift:.4e} new_shift = {:.4e}",
463                            self.data.shifts[i]);
464                //}}}
465            }
466        }
467    }
468    //}}}
469}

566//{{{ struct: AugmentedLagrangianFcn
567#[derive(Debug)]
568pub struct AugmentedLagrangianFcn<F1: RealFn, F2: RealVectorFn, F3: RealVectorFn> {
569    fcn: F1,
570    eq_penalty: Option<EqPenalty<F2>>,
571    ieq_penalty: Option<IeqPenalty<F3>>,
572    lagrangian_type: LagrangianType,
573    cached_values: HashMap<LagrangianType, CachedValues>,
574}
575//}}}
576//{{{ impl: AugmentedLagrangianFcn
577impl<F1: RealFn, F2: RealVectorFn, F3: RealVectorFn> AugmentedLagrangianFcn<F1, F2, F3> {
578    //{{{ fn: new
579    #[trace_fn]
580    pub fn new(
581        fcn: F1,
582        eq_constraints: Option<F2>,
583        ieq_constraints: Option<F3>,
584        initial_penalty: f64,
585        lagrangian_type: LagrangianType,
586    ) -> Self {
587        let eq_penalty =
588            eq_constraints.map(|eq_con| EqPenalty::new(eq_con, initial_penalty, lagrangian_type));
589
590        let ieq_penalty = ieq_constraints
591            .map(|ieq_con| IeqPenalty::new(ieq_con, initial_penalty, lagrangian_type));
592
593        let n = fcn.dimension_domain();
594        let mut cached_values = HashMap::<LagrangianType, CachedValues>::new();
595        cached_values.insert(LagrangianType::AugmentedLagrangian, CachedValues::new(n));
596        cached_values.insert(LagrangianType::Lagrangian, CachedValues::new(n));
597
598        Self {
599            fcn,
600            eq_penalty,
601            ieq_penalty,
602            lagrangian_type,
603            cached_values,
604        }
605    }
606    //}}}
607    //{{{ fn: set_lagrangian_type
608    #[trace_fn]
609    fn set_lagrangian_type(
610        &mut self,
611        lagrangian_type: LagrangianType,
612    ) {
613        self.lagrangian_type = lagrangian_type;
614        if let Some(eq_penalty) = &mut self.eq_penalty {
615            eq_penalty.lagrangian_type = lagrangian_type
616        }
617
618        if let Some(ieq_penalty) = &mut self.ieq_penalty {
619            ieq_penalty.lagrangian_type = lagrangian_type
620        }
621    }
622    //}}}
623    fn evaluate_for_type(
624        &mut self,
625        x: &Vector,
626        lagrangian_type: LagrangianType,
627    ) {
628        let previous_type = self.lagrangian_type;
629        self.set_lagrangian_type(lagrangian_type);
630        let _ = self.eval(x);
631        let _ = self.derivative(x);
632        self.set_lagrangian_type(previous_type);
633    }
634    //{{{ fn: update_max_eq_violations
635    #[trace_fn]
636    fn update_max_eq_violations(
637        &self,
638        max_eq_violations: &mut Vector,
639    ) {
640        let eq_penalty = self.eq_penalty.as_ref().unwrap();
641
642        for (max_violation, hi) in max_eq_violations
643            .iter_mut()
644            .zip(eq_penalty.data.values.iter())
645        {
646            *max_violation = max_violation.max(hi.abs());
647        }
648    }
649    //}}}
650    //{{{ fn: update_max_ieq_violations
651    #[trace_fn]
652    fn update_max_ieq_violations(
653        &self,
654        max_ieq_violations: &mut Vector,
655    ) {
656        let ieq_penalty = self.ieq_penalty.as_ref().unwrap();
657
658        for (max_violation, gi, shift_i) in max_ieq_violations
659            .iter_mut()
660            .zip(ieq_penalty.data.values.iter())
661            .zip(ieq_penalty.data.shifts.iter())
662            .map(|((max_violation, gi), phi_i)| (max_violation, gi, phi_i))
663        {
664            *max_violation = max_violation.max(((-gi).min(*shift_i)).abs());
665        }
666    }
667    //}}}
668    //{{{ fn: has_eq_penalty
669    #[trace_fn]
670    fn has_eq_penalty(&self) -> bool {
671        self.eq_penalty.is_some()
672    }
673    //}}}
674    //{{{ fn: has_ieq_penalty
675    #[trace_fn]
676    fn has_ieq_penalty(&self) -> bool {
677        self.ieq_penalty.is_some()
678    }
679    //}}}
680    //{{{ fn: get_cached_values
681    #[trace_fn]
682    fn get_cached_values(&self) -> &CachedValues {
683        self.cached_values.get(&self.lagrangian_type).unwrap()
684    }
685    //}}}
686    //{{{ fn: get_cached_values_mut
687    #[trace_fn]
688    fn get_cached_values_mut(&mut self) -> &mut CachedValues {
689        self.cached_values.get_mut(&self.lagrangian_type).unwrap()
690    }
691    //}}}
692    //{{{ fn: is_constrained
693    #[trace_fn]
694    fn is_constrained(&self) -> bool {
695        self.has_eq_penalty() || self.has_ieq_penalty()
696    }
697    //}}}
698}

719    //{{{ fn: eval
720    #[trace_fn]
721    fn eval(
722        &mut self,
723        x: &Vector,
724    ) -> f64 {
725        let fcn_value = self.fcn.eval(x);
726
727        //{{{ trace
728        info!(target: "aug", "Evaluated objective: {fcn_value:.4e}");
729        //}}}
730
731        let eq_value = if let Some(eq_penalty) = &mut self.eq_penalty {
732            let eq_value = eq_penalty.eval(x);
733            //{{{ trace
734            debug!(target: "aug", "Evaluated EQ penalty: {eq_value:.4e}");
735            //}}}
736            eq_value
737        } else {
738            0.0
739        };
740
741        let ieq_value = if let Some(ieq_penalty) = &mut self.ieq_penalty {
742            let ieq_value = ieq_penalty.eval(x);
743            //{{{ trace
744            debug!(target: "aug", "Evaluated IEQ penalty: {ieq_value:.4e}");
745            //}}}
746            ieq_value
747        } else {
748            0.0
749        };
750
751        let auglag_value = fcn_value + eq_value + ieq_value;
752
753        {
754            let cached_value = self.get_cached_values_mut();
755            cached_value.fcn_value = fcn_value;
756            cached_value.eq_constraint_value = eq_value;
757            cached_value.ieq_constraint_value = ieq_value;
758            cached_value.all_constraint_value = eq_value + ieq_value;
759            cached_value.auglag_value = auglag_value;
760        }
761
762        //{{{ trace
763        debug!(target: "aug", "Augmented Lagrangian value: {auglag_value:.4e}");
764        //}}}
765        auglag_value
766    }

768    //{{{ fn: grad
769    #[trace_fn]
770    fn derivative(
771        &mut self,
772        x: &Vector,
773    ) -> Vector {
774        let n = self.dimension_domain();
775
776        let fcn_grad = self.fcn.derivative(x);
777
778        //{{{ trace
779        trace!(target: "aug", "Evaluated Objective Gradient = {}",
780                    fcn_grad.clone().transpose());
781        //}}}
782
783        let eq_penalty_grad = if let Some(eq_constraint_data) = &mut self.eq_penalty {
784            let eq_penalty_grad = eq_constraint_data.derivative(x);
785            //{{{ trace
786            trace!(target: "aug", "Evaluated EQ Gradient = {}",
787                    eq_penalty_grad.clone().transpose());
788            //}}}
789            eq_penalty_grad
790        } else {
791            Vector::zeros_vec(n, Col)
792        };
793
794        let ieq_penalty_grad = if let Some(ieq_constraint_data) = &mut self.ieq_penalty {
795            let ieq_penalty_grad = ieq_constraint_data.derivative(x);
796            //{{{ trace
797            trace!(target: "aug", "Evaluated IEQ Gradient = {}",
798                    ieq_penalty_grad.clone().transpose());
799            //}}}
800            ieq_penalty_grad
801        } else {
802            Vector::zeros_vec(n, Col)
803        };
804
805        let grad_aug_lag: Vector = (&fcn_grad + &eq_penalty_grad + &ieq_penalty_grad).into();
806
807        {
808            let cached_value = self.get_cached_values_mut();
809            cached_value.fcn_grad = fcn_grad;
810            cached_value.eq_constraint_grad = eq_penalty_grad.clone();
811            cached_value.ieq_constraint_grad = ieq_penalty_grad.clone();
812            cached_value.all_constraint_grad = (&eq_penalty_grad + &ieq_penalty_grad).into();
813            cached_value.auglag_grad = grad_aug_lag.clone();
814        }
815
816        //{{{ trace
817        trace!(target: "aug", "Evaluated Augmented Lagrangian Gradient = {}",
818                    grad_aug_lag.clone().transpose());
819        //}}}
820        grad_aug_lag
821    }

823}

826//{{{ struct: AugmentedLagrangian
827pub struct AugmentedLagrangian<F1: RealFn, F2: RealVectorFn, F3: RealVectorFn> {
828    fcn: AugmentedLagrangianFcn<F1, F2, F3>,
829    x_init: Vector,
830    bounds: Option<BoundConstraints>,
831    opts: Options,
832    omega_k: f64,
833    /// Scheduling-only ramp used by `compute_omega_k`; grows every outer
834    /// iteration unconditionally. Deliberately decoupled from the real
835    /// per-constraint penalties (`EqPenalty`/`IeqPenalty`), which only grow
836    /// when a constraint fails to improve and can otherwise stay fixed
837    /// forever — seeding ω_k's floor off the real penalty would then let it
838    /// get stuck loose indefinitely.
839    mu_k: f64,
840}
841//}}}
842//{{{ impl: AugmentedLagrangian
843impl<F1: RealFn, F2: RealVectorFn, F3: RealVectorFn> AugmentedLagrangian<F1, F2, F3> {
844    //{{{ fn: new
845    #[trace_fn]
846    pub fn new(
847        fcn: F1,
848        bounds: Option<BoundConstraints>,
849        eq_constraints: Option<F2>,
850        ieq_constraints: Option<F3>,
851        x0: Vector,
852        opts: Options,
853    ) -> Self {
854        //{{{ trace
855        trace!(target: "aug", "Creating new Augmented Lagrangian function");
856        //}}}
857
858        let mut fcn = AugmentedLagrangianFcn::new(
859            fcn,
860            eq_constraints,
861            ieq_constraints,
862            opts.initial_penalty,
863            LagrangianType::AugmentedLagrangian,
864        );
865
866        let _ = fcn.eval(&x0);
867        let _ = fcn.derivative(&x0);
868
869        let user_omega = opts.constrained_opts.base_opts.grad_atol;
870        let mu0 = opts.initial_penalty;
871        Self {
872            fcn,
873            x_init: x0,
874            bounds,
875            opts,
876            omega_k: OMEGA_INIT_CEIL.max(user_omega),
877            mu_k: mu0,
878        }
879    }

881    //{{{ fn: is_converged
882    #[trace_fn]
883    fn is_converged(
884        &mut self,
885        iter_k: &IterData,
886    ) -> Option<ConvergedReason> {
887        //{{{ trace
888        info!(target: "aug", "Checking convergence");
889        //}}}
890        self.fcn
891            .evaluate_for_type(&iter_k.x, LagrangianType::Lagrangian);
892
893        let norm_eq = if let Some(eq_penalty) = &self.fcn.eq_penalty {
894            eq_penalty.data.values.abs_max().unwrap()
895        } else {
896            0.0
897        };
898        let norm_ieq = if let Some(ieq_penalty) = &self.fcn.ieq_penalty {
899            ieq_penalty.data.values.posed().abs_max().unwrap()
900        } else {
901            0.0
902        };
903
904        let cached_values = self
905            .fcn
906            .cached_values
907            .get(&LagrangianType::Lagrangian)
908            .unwrap();
909        let norm_grad_f = cached_values.fcn_grad.abs_max().unwrap();
910        let norm_grad_penalty = cached_values.all_constraint_grad.abs_max().unwrap();
911        let stationarity_gradient = if let Some(bounds) = &self.bounds {
912            bounds.projected_direction(&iter_k.x, &(-cached_values.auglag_grad.clone()), 1.0)
913        } else {
914            cached_values.auglag_grad.clone()
915        };
916        let _norm_grad_auglag = cached_values.auglag_grad.abs_max().unwrap();
917        let norm_projected_grad_auglag = stationarity_gradient.abs_max().unwrap_or(0.0);
918
919        let residual_stationarity = norm_projected_grad_auglag;
920        let residual_stationarity_scaled =
921            norm_projected_grad_auglag / 1.0f64.max(norm_grad_f).max(norm_grad_penalty);
922        let residual_primal = norm_eq.max(norm_ieq);
923        //{{{ trace
924        info!(target: "aug", "||∇P|| = {norm_grad_penalty:.4e} ||∇F|| = {norm_grad_f:.4} ||h|| = {norm_eq:.4e} ||g|| = {norm_ieq:.4e}");
925        info!(target: "aug", "||∇L|| = {_norm_grad_auglag:.4e})");
926        info!(target: "aug", "||∇L_proj|| = {norm_projected_grad_auglag:.4e})");
927        info!(target: "aug", "||∇L_proj|| / max(1, ||∇F||, ||∇P||) = {residual_stationarity_scaled:.4e}");
928        //}}}
929        let ctol = self.opts.constrained_opts.constraint_tol;
930        let rtol = self.opts.constrained_opts.base_opts.grad_rtol;
931        let atol = self.opts.constrained_opts.base_opts.grad_atol;
932        let constraints_satsifed = residual_primal < ctol;
933        let stationarity_rtol_satisfied = residual_stationarity_scaled < rtol;
934        let stationarity_atol_satisfied = residual_stationarity < atol;
935
936        if constraints_satsifed && stationarity_rtol_satisfied {
937            Some(ConvergedReason::Rtol)
938        } else if constraints_satsifed && stationarity_atol_satisfied {
939            Some(ConvergedReason::Atol)
940        } else {
941            None
942        }
943    }
944    //}}}

970    //{{{ fn: update_penalties_shifts
971    #[trace_fn]
972    fn update_lagrangian(
973        &mut self,
974        constraint_improvement_factor: f64,
975        penalty_increase_factor: f64,
976        uncon_ret: VectorReturns,
977    ) -> IterData {
978        if let Some(eq_penalty) = &mut self.fcn.eq_penalty {
979            //{{{ trace
980            info!(target: "aug", "Updating penalties and shifts for EQ");
981            //}}}
982            eq_penalty
983                .update_penalties_shifts(constraint_improvement_factor, penalty_increase_factor);
984        }
985
986        if let Some(ieq_penalty) = &mut self.fcn.ieq_penalty {
987            //{{{ trace
988            info!(target: "aug", "Updating penalties and shifts for IEQ");
989            //}}}

996    /// Computes this outer iteration's inner stationarity tolerance ω_k and
997    /// advances the stored ceiling for the next call.
998    ///
999    /// `ω_k` follows the classical LANCELOT/Birgin–Martínez ramp
1000    /// `0.1 * (1/μ_k)^1.5`, clamped to `[user_grad_atol, previous ω_k]`. That
1001    /// makes the sequence monotone non-increasing (never loosens between
1002    /// outer iterations), reach tolerances tighter than the old hardcoded
1003    /// 1e-6 floor when the user asks for them via `grad_atol`, and — because
1004    /// `μ_k` (see the `mu_k` field) grows by a fixed factor every outer
1005    /// iteration unconditionally — land on `user_grad_atol` within a bounded
1006    /// number of iterations regardless of problem-specific behavior.
1007    ///
1008    /// Earlier versions scaled this off `max(‖c‖∞, 1/μ_k)`, tying it to the
1009    /// live constraint residual. Two failure modes ruled that out:
1010    /// - Seeding the floor off the *real* per-constraint penalty stalled
1011    ///   forever, since `update_max_violations` only grows a constraint's
1012    ///   penalty when that constraint fails to improve — one that's already
1013    ///   comfortably satisfied can leave its penalty fixed indefinitely.
1014    /// - Seeding it off the live residual instead (even via a scheduling-only
1015    ///   `μ_k`) let `ω_k` keep chasing the residual down long after outer
1016    ///   stationarity was already satisfied and only feasibility was still
1017    ///   converging (which happens on its own schedule via the multiplier
1018    ///   update, not by solving the inner problem tighter) — for
1019    ///   ill-conditioned inner subproblems (e.g. quartics, or gradients with
1020    ///   O(10) magnitude requiring near-cancellation to resolve an O(1e-8)
1021    ///   residual) this pushed past the inner solver's achievable precision
1022    ///   and the line search failed outright. A pure μ_k ramp, independent
1023    ///   of the live residual, avoids both.
1024    #[trace_fn]
1025    fn compute_omega_k(&mut self) -> f64 {
1026        let user_omega = self.opts.constrained_opts.base_opts.grad_atol;
1027        let prev_omega_k = self.omega_k;
1028        let mu_k = self.mu_k;
1029        self.mu_k *= self.opts.penalty_growth_factor;
1030
1031        let omega_target = 0.1 * (1.0 / mu_k).powf(1.5);
1032        let omega_k = omega_target.max(user_omega).min(prev_omega_k);
1033
1034        self.omega_k = omega_k;
1035        //{{{ trace
1036        info!(target: "aug", "Inner ω_k = {omega_k:.4e} (prev = {prev_omega_k:.4e}, user floor = {user_omega:.4e})");
1037        //}}}
1038        omega_k
1039    }
1040
1041    #[trace_fn]
1042    fn set_inner_tolerances(&mut self) -> InnerMethod {
1043        let inner_method = self.opts.inner_method.clone();
1044        let is_constrained = self.fcn.is_constrained();
1045
1046        match inner_method {
1047            InnerMethod::Unconstrained(mut uncon_method) => {
1048                if !is_constrained {
1049                    uncon_method.uncon_opts_mut().grad_rtol =
1050                        self.opts.constrained_opts.base_opts.grad_rtol;
1051                    uncon_method.uncon_opts_mut().grad_atol =
1052                        self.opts.constrained_opts.base_opts.grad_atol;
1053                } else {
1054                    let omega_k = self.compute_omega_k();
1055                    uncon_method.uncon_opts_mut().grad_rtol = 0.0;
1056                    uncon_method.uncon_opts_mut().grad_atol = omega_k;
1057                }
1058                InnerMethod::Unconstrained(uncon_method)
1059            }
1060            InnerMethod::BoundConstrained(mut bcon_method) => {
1061                if !is_constrained {
1062                    bcon_method.bound_opts_mut().base_opts.grad_rtol =
1063                        self.opts.constrained_opts.base_opts.grad_rtol;
1064                    bcon_method.bound_opts_mut().base_opts.grad_atol =
1065                        self.opts.constrained_opts.base_opts.grad_atol;
1066                } else {
1067                    let omega_k = self.compute_omega_k();
1068                    bcon_method.bound_opts_mut().base_opts.grad_rtol = 0.0;
1069                    bcon_method.bound_opts_mut().base_opts.grad_atol = omega_k;
1070                }
1071                InnerMethod::BoundConstrained(bcon_method)
1072            }
1073        }
1074    }
1075}

1078/// Selects the optimizer used for an augmented-Lagrangian inner problem.
1079#[cfg_attr(feature = "serde", derive(serde::Deserialize, serde::Serialize))]
1080#[derive(Clone, Debug, PartialEq)]
1081#[non_exhaustive]
1082pub enum InnerMethod {
1083    /// Use an unconstrained inner optimizer.
1084    Unconstrained(UnconstrainedMethod),
1085    /// Use a bound-constrained inner optimizer.
1086    BoundConstrained(BoundConstrainedMethod),
1087}

1091    ///
1092    /// # Errors

1098            Self::BoundConstrained(method) => method.validate(),

1102//{{{ fn: inner_minimize
1103fn inner_minimize<F: RealFn + ?Sized>(
1104    fcn: &mut F,
1105    bounds: Option<BoundConstraints>,
1106    x0: Vector,
1107    inner_method: InnerMethod,
1108) -> Result<common::VectorReturns, ConstrainedError> {
1109    match inner_method {
1110        InnerMethod::Unconstrained(uncon_method) => Ok(uncon_minimize(fcn, x0, uncon_method)?),
1111        InnerMethod::BoundConstrained(bcon_method) => {
1112            Ok(bcon_minimize(fcn, bounds.unwrap(), x0, bcon_method)?)
1113        }
1114    }
1115}

1117//{{{ impl: Minimizer for AugmentedLagrangian
1118impl<F1: RealFn, F2: RealVectorFn, F3: RealVectorFn> Minimizer for AugmentedLagrangian<F1, F2, F3> {
1119    type Error = ConstrainedError;
1120    type Returns = VectorReturns;
1121
1122    #[trace_fn]
1123    fn minimize(&mut self) -> Result<VectorReturns, Self::Error> {
1124        let alpha = self.opts.constraint_improvement_factor;
1125        let beta = self.opts.penalty_growth_factor;
1126        let n_iter = self.opts.constrained_opts.base_opts.max_iter;
1127        let mut iter_k = IterData::new(&mut self.fcn, &self.x_init);
1128        let mut iter_prev_k = iter_k.clone();
1129
1130        for k in 1..n_iter {
1131            self.print_status(k, &iter_k);
1132            let uncon_method = self.set_inner_tolerances();
1133            let _omega_k = self.omega_k;
1134            let ret = match inner_minimize(
1135                &mut self.fcn,
1136                self.bounds.clone(),
1137                iter_prev_k.x.clone(),
1138                uncon_method,
1139            ) {
1140                Ok(ret) => ret,
1141                // A line-search failure here (typically "no decreasing step found")
1142                // usually means the inner iterate is already stationary to numerical
1143                // precision for the requested ω_k — the augmented Lagrangian's
1144                // penalty-gradient term can dominate the objective's own gradient
1145                // near the optimum, so resolving a further decrease runs into
1146                // floating-point cancellation before ω_k reaches its target. Treat
1147                // the unmoved iterate as this step's result (the outer `is_converged`
1148                // check still gates whether that's actually good enough) instead of
1149                // aborting the whole outer solve; if the inner solver is genuinely
1150                // stuck rather than just precision-limited, this degrades to
1151                // `MaxIterations` once the outer loop stops making progress, rather
1152                // than masking a real failure as success.
1153                Err(_err) => {
1154                    //{{{ trace
1155                    info!(target: "aug", "Inner solve failed at ω_k = {_omega_k:.4e}: {_err:?} — treating {k}'s starting iterate as this step's result");
1156                    //}}}
1157                    let fmin = self.fcn.fcn.eval(&iter_prev_k.x);
1158                    VectorReturns {
1159                        xmin: iter_prev_k.x.clone(),
1160                        fmin,
1161                        reason: ConvergedReason::Atol,
1162                        num_iterations: 0,
1163                        num_fun_evals: 0,
1164                        num_grad_evals: 0,
1165                    }
1166                }
1167            };
1168            iter_k = self.update_lagrangian(alpha, beta, ret);
1169            iter_prev_k.copy_from(&iter_k);
1170            if let Some(reason) = self.is_converged(&iter_k) {
1171                //{{{ trace
1172                info!(target: "cg", "*********************************************");
1173                info!(target: "cg", "Converging with reason {reason:?}");
1174                info!(target: "cg","Convergence measures:");
1175                info!(target: "cg", "||∇L(k)|| = {:.4e}", iter_k.norm_grad_fx);
1176                info!(target: "cg", "*********************************************");
1177                //}}}
1178                let fmin = self.fcn.fcn.eval(&iter_k.x);
1179                let xmin = iter_k.x;
1180                return Ok(VectorReturns {
1181                    fmin,
1182                    xmin,
1183                    reason,
1184                    num_iterations: k as usize,
1185                    num_fun_evals: 0,
1186                    num_grad_evals: 0,
1187                });
1188            }
1189        }
1190
1191        Err(ConstrainedError::MaxIterations(0_usize))
1192    }
1193}
