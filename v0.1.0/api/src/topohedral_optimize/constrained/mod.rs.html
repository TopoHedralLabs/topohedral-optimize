38//{{{ fn: minimize
39#[trace_fn]
40/// Minimizes a function subject to optional bounds and vector constraints.
41///
42/// Equality constraints are interpreted as `h(x) = 0`; inequality constraints
43/// are interpreted as `g(x) <= 0`.
44///
45/// # Errors
46///
47/// Returns [`ConstrainedError`] if method options or dimensions are invalid,
48/// bounds are paired with an incompatible inner solver, an inner solve fails,
49/// or the iteration limit is reached.
50pub fn minimize<F: RealFn + ?Sized>(
51    fcn: &mut F,
52    bounds: Option<BoundConstraints>,
53    eq_constraints: Option<&mut dyn RealVectorFn>,
54    ieq_constraints: Option<&mut dyn RealVectorFn>,
55    x0: Vector,
56    method: ConstrainedMethod,
57) -> Result<VectorReturns, ConstrainedError> {
58    let mut evaluator = Evaluator::new(fcn);
59    let mut ret = minimize_impl(
60        &mut evaluator,
61        bounds,
62        eq_constraints,
63        ieq_constraints,
64        x0,
65        method,
66    )?;
67    ret.num_fun_evals = evaluator.num_func_evals;
68    ret.num_grad_evals = evaluator.num_grad_evals;
69    Ok(ret)
70}
71
72pub(crate) fn minimize_impl<F: RealFn + ?Sized>(
73    fcn: &mut F,
74    bounds: Option<BoundConstraints>,
75    eq_constraints: Option<&mut dyn RealVectorFn>,
76    ieq_constraints: Option<&mut dyn RealVectorFn>,
77    x0: Vector,
78    method: ConstrainedMethod,
79) -> Result<VectorReturns, ConstrainedError> {
80    method.validate()?;
81    let expected = fcn.dimension_domain();
82    if x0.len() != expected {
83        return Err(ValidationError::DimensionMismatch {
84            parameter: "x0",
85            expected,
86            actual: x0.len(),
87        }
88        .into());
89    }
90    if let Some(bounds) = bounds.as_ref() {
91        let actual = bounds.dimension_domain();
92        if actual != expected {
93            return Err(ValidationError::DimensionMismatch {
94                parameter: "bounds",
95                expected,
96                actual,
97            }
98            .into());
99        }
100    }
101    if let Some(constraints) = eq_constraints.as_ref() {
102        let actual = constraints.dimension_domain();
103        if actual != expected {
104            return Err(ValidationError::DimensionMismatch {
105                parameter: "eq_constraints",
106                expected,
107                actual,
108            }
109            .into());
110        }
111    }
112    if let Some(constraints) = ieq_constraints.as_ref() {
113        let actual = constraints.dimension_domain();
114        if actual != expected {
115            return Err(ValidationError::DimensionMismatch {
116                parameter: "ieq_constraints",
117                expected,
118                actual,
119            }
120            .into());
121        }
122    }
123    let minimizer = factory::create(fcn, bounds, eq_constraints, ieq_constraints, x0, method);
124    minimizer?.minimize()
125}
